\subsubsection*{Concepto clave 3: Modelo de pérdida de trayectoria (Path Loss)}
\textbf{Fórmula teórica (espacio libre):}

\[
PL(dB)=20\log_{10}(d)+20\log_{10}(f)+32.44
\]

\noindent donde $d$ = distancia (km), $f$ = frecuencia (MHz).

\textbf{Cálculo para mi simulación:}

\begin{itemize}[leftmargin=1.2cm]
    \item Distancia 20 m = 0.02 km
    \item Frecuencia = 2450 MHz
    \item $PL = 20\log_{10}(0.02) + 20\log_{10}(2450) + 32.44 = \underline{\hspace{2cm}}\; dB$
\end{itemize}
